# GPU driver (Intel i915, DRM/KMS). IMPORTANT: do NOT install via the
# `drm-kmod` meta-port on this unit -- it pulls drm-612-kmod, which panics
# the kernel on this GPU (Sandy Bridge, PCI 8086:0126; confirmed open
# upstream bug github.com/freebsd/drm-kmod#526). Install the 6.1 branch
# directly instead:
#   pkg install -y drm-61-kmod
# and set kld_list="i915kms" in /etc/rc.conf.

# Gotchas found during setup:
# - openjdk25 (pulled in by maven/leiningen) needs procfs mounted:
#     mount -t procfs proc /proc   (and an /etc/fstab line for persistence -- done)
# - qt5ct needs QT_QPA_PLATFORMTHEME=qt5ct exported (set in i3/xinitrc)
# - FreeBSD's `install -D` does NOT auto-create parent dirs like GNU install's
#   does -- install.sh's firefox policy step does `mkdir -p` first explicitly.

# "MesloLGS NF" (exact family name powerlevel10k/Xresources/i3 reference) has
# no FreeBSD package -- nerd-fonts-meslo uses different family names, which
# silently breaks powerline glyphs. Fetched directly from
# romkatv/powerlevel10k-media into ~/.local/share/fonts (now scripted in
# install.sh).
